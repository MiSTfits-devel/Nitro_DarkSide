module tiny_audio
#(
	parameter CLOCK_RATE_HZ = 24576000
)
(
	input                clk,
	input                reset,

	input                sample_rate_is_96khz,

	input                mute,
	input          [3:0] attenuation_shift,

	input  signed [15:0] audio_in_left,
	input  signed [15:0] audio_in_right,

	output               i2s_bit_clock,
	output               i2s_frame_clock,
	output               i2s_serial_data,

	output               spdif_out,

	output               dac_pulse_left,
	output               dac_pulse_right
);

localparam [31:0] AUDIO_SAMPLE_RATE_48K_HZ = 32'd48000;

wire [31:0] audio_sample_rate_hz = sample_rate_is_96khz ? (AUDIO_SAMPLE_RATE_48K_HZ << 1) : AUDIO_SAMPLE_RATE_48K_HZ;

wire audio_sample_ce;
tiny_audio_rate_generator #(.CLOCK_RATE_HZ(CLOCK_RATE_HZ)) sample_rate_generator
(
	.clk(clk),
	.reset(reset),
	.target_rate_hz(audio_sample_rate_hz),
	.pulse_out(audio_sample_ce)
);

wire i2s_bit_toggle_ce;
tiny_audio_rate_generator #(.CLOCK_RATE_HZ(CLOCK_RATE_HZ)) i2s_rate_generator
(
	.clk(clk),
	.reset(reset),
	.target_rate_hz(audio_sample_rate_hz << 6),
	.pulse_out(i2s_bit_toggle_ce)
);

wire spdif_half_bit_ce;
tiny_audio_rate_generator #(.CLOCK_RATE_HZ(CLOCK_RATE_HZ)) spdif_rate_generator
(
	.clk(clk),
	.reset(reset),
	.target_rate_hz(audio_sample_rate_hz << 7),
	.pulse_out(spdif_half_bit_ce)
);

reg [19:0] startup_mute_counter = 0;
wire       startup_mute = ~startup_mute_counter[19];
always @(posedge clk) begin
	if(reset) startup_mute_counter <= 0;
	else if(audio_sample_ce && !startup_mute_counter[19]) startup_mute_counter <= startup_mute_counter + 1'd1;
end

wire combined_mute = mute | startup_mute;

wire signed [15:0] filtered_left, filtered_right;
tiny_audio_dc_blocker dc_blocker_left
(
	.clk(clk),
	.ce(audio_sample_ce),
	.sample_rate_is_96khz(sample_rate_is_96khz),
	.sample_in(audio_in_left),
	.sample_out(filtered_left)
);

tiny_audio_dc_blocker dc_blocker_right
(
	.clk(clk),
	.ce(audio_sample_ce),
	.sample_rate_is_96khz(sample_rate_is_96khz),
	.sample_in(audio_in_right),
	.sample_out(filtered_right)
);

wire signed [15:0] attenuated_left  = combined_mute ? 16'sd0 : (filtered_left  >>> attenuation_shift);
wire signed [15:0] attenuated_right = combined_mute ? 16'sd0 : (filtered_right >>> attenuation_shift);

tiny_audio_i2s_transmitter i2s_transmitter
(
	.clk(clk),
	.reset(reset),
	.bit_toggle_ce(i2s_bit_toggle_ce),
	.left_sample(attenuated_left),
	.right_sample(attenuated_right),
	.bit_clock(i2s_bit_clock),
	.frame_clock(i2s_frame_clock),
	.serial_data(i2s_serial_data)
);

tiny_audio_spdif_transmitter spdif_transmitter
(
	.clk(clk),
	.reset(reset),
	.half_bit_ce(spdif_half_bit_ce),
	.left_sample(attenuated_left),
	.right_sample(attenuated_right),
	.spdif_out(spdif_out)
);

wire [15:0] offset_binary_left  = {~attenuated_left[15],  attenuated_left[14:0]};
wire [15:0] offset_binary_right = {~attenuated_right[15], attenuated_right[14:0]};

tiny_audio_sigma_delta_dac dac_left
(
	.clk(clk),
	.reset(reset),
	.offset_binary_sample(offset_binary_left),
	.pulse_out(dac_pulse_left)
);

tiny_audio_sigma_delta_dac dac_right
(
	.clk(clk),
	.reset(reset),
	.offset_binary_sample(offset_binary_right),
	.pulse_out(dac_pulse_right)
);

endmodule

module tiny_audio_rate_generator
#(
	parameter CLOCK_RATE_HZ = 24576000
)
(
	input             clk,
	input             reset,
	input      [31:0] target_rate_hz,
	output reg        pulse_out = 0
);

reg [31:0] phase_accumulator = 0;

always @(posedge clk) begin
	reg [31:0] next_phase_accumulator;

	if(reset) begin
		phase_accumulator <= 0;
		pulse_out         <= 0;
	end
	else begin
		next_phase_accumulator = phase_accumulator + target_rate_hz;
		pulse_out = 1'b0;

		if(next_phase_accumulator >= CLOCK_RATE_HZ) begin
			next_phase_accumulator = next_phase_accumulator - CLOCK_RATE_HZ;
			pulse_out = 1'b1;
		end

		phase_accumulator <= next_phase_accumulator;
	end
end

endmodule

module tiny_audio_dc_blocker
(
	input                clk,
	input                ce,
	input                sample_rate_is_96khz,
	input  signed [15:0] sample_in,
	output signed [15:0] sample_out
);

reg signed [31:0] previous_input_extended = 0;
reg signed [31:0] filter_accumulator = 0;

always @(posedge clk) if(ce) begin
	reg signed [31:0] input_extended;
	reg signed [31:0] leak_term;

	input_extended = {{16{sample_in[15]}}, sample_in};
	leak_term = sample_rate_is_96khz ? (filter_accumulator >>> 13) : (filter_accumulator >>> 12);

	filter_accumulator      <= filter_accumulator + input_extended - previous_input_extended - leak_term;
	previous_input_extended <= input_extended;
end

assign sample_out =
	(filter_accumulator[31:15] != {17{filter_accumulator[15]}}) ?
		(filter_accumulator[31] ? 16'sh8000 : 16'sh7FFF) :
		filter_accumulator[15:0];

endmodule

module tiny_audio_i2s_transmitter
(
	input         clk,
	input         reset,
	input         bit_toggle_ce,
	input  [15:0] left_sample,
	input  [15:0] right_sample,
	output reg    bit_clock              = 1,
	output reg    frame_clock            = 1,
	output reg    serial_data            = 0
);

reg [15:0] shift_register         = 0;
reg  [4:0] bits_remaining_in_word = 0;

always @(posedge clk) begin
	if(reset) begin
		bit_clock              <= 1'b1;
		frame_clock            <= 1'b1;
		bits_remaining_in_word <= 5'd15;
		shift_register         <= right_sample;
	end
	else if(bit_toggle_ce) begin
		bit_clock <= ~bit_clock;

		if(bit_clock) begin
			serial_data <= shift_register[15];

			if(bits_remaining_in_word == 0) begin
				bits_remaining_in_word <= 5'd15;
				frame_clock            <= ~frame_clock;
				shift_register         <= frame_clock ? left_sample : right_sample;
			end
			else begin
				bits_remaining_in_word <= bits_remaining_in_word - 1'd1;
				shift_register         <= {shift_register[14:0], 1'b0};
			end
		end
	end
end

endmodule

module tiny_audio_sigma_delta_dac
(
	input         clk,
	input         reset,
	input  [15:0] offset_binary_sample,
	output reg    pulse_out = 0
);

reg [16:0] error_accumulator = 0;

always @(posedge clk) begin
	if(reset) begin
		error_accumulator <= 0;
		pulse_out         <= 0;
	end
	else begin
		error_accumulator <= error_accumulator[15:0] + offset_binary_sample;
		pulse_out         <= error_accumulator[16];
	end
end

endmodule

module tiny_audio_spdif_transmitter
(
	input                clk,
	input                reset,
	input                half_bit_ce,
	input  signed [15:0] left_sample,
	input  signed [15:0] right_sample,
	output reg           spdif_out = 0
);

localparam [7:0] PREAMBLE_BLOCK_START   = 8'b11101000;
localparam [7:0] PREAMBLE_LEFT_CHANNEL  = 8'b11100010;
localparam [7:0] PREAMBLE_RIGHT_CHANNEL = 8'b11100100;

reg [8:0] subframe_index_in_block    = 0;
reg [5:0] half_bit_index_in_subframe = 0;

wire subframe_is_right_channel = subframe_index_in_block[0];
wire block_is_starting         = (subframe_index_in_block == 0);

wire [7:0] active_preamble =
	block_is_starting         ? PREAMBLE_BLOCK_START :
	subframe_is_right_channel ? PREAMBLE_RIGHT_CHANNEL :
	                             PREAMBLE_LEFT_CHANNEL;

wire signed [15:0] active_sample = subframe_is_right_channel ? right_sample : left_sample;

wire [26:0] data_cells_before_parity = {1'b0, 1'b0, 1'b0, active_sample, 8'd0};
wire        parity_cell              = ^data_cells_before_parity;
wire [27:0] data_cells               = {parity_cell, data_cells_before_parity};

always @(posedge clk) begin
	if(reset) begin
		subframe_index_in_block    <= 0;
		half_bit_index_in_subframe <= 0;
		spdif_out                  <= 0;
	end
	else if(half_bit_ce) begin
		if(half_bit_index_in_subframe < 6'd8) begin
			spdif_out <= active_preamble[7 - half_bit_index_in_subframe];
		end
		else if(half_bit_index_in_subframe[0] == 1'b0) begin
			spdif_out <= ~spdif_out;
		end
		else if(data_cells[(half_bit_index_in_subframe - 6'd8) >> 1]) begin
			spdif_out <= ~spdif_out;
		end

		if(half_bit_index_in_subframe == 6'd63) begin
			half_bit_index_in_subframe <= 0;
			subframe_index_in_block    <= (subframe_index_in_block == 9'd383) ? 9'd0 : subframe_index_in_block + 9'd1;
		end
		else begin
			half_bit_index_in_subframe <= half_bit_index_in_subframe + 6'd1;
		end
	end
end

endmodule
